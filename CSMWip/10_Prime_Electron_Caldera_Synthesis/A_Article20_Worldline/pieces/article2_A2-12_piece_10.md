import numpy as np
from mpmath import zeta

class BostConnesSystem:
    def __init__(self, max_n=1000):
        self.max_n = max_n
        self.H = np.diag([np.log(n) for n in range(1, max_n+1)])
    
    def partition_function(self, beta):
        return zeta(beta)
    
    def kms_state(self, beta):
        Z = self.partition_function(beta)
        if Z == np.inf or Z <= 0:
            return None
        rho = np.diag([n**(-beta) / Z for n in range(1, self.max_n+1)])
        return rho
    
    def free_energy(self, beta):
        Z = self.partition_function(beta)
        return -np.log(Z) / beta
    
    def specific_heat(self, beta, h=1e-4):
        F1 = self.free_energy(beta + h)
        F2 = self.free_energy(beta - h)
        F0 = self.free_energy(beta)
        return -beta**2 * (F1 - 2*F0 + F2) / h**2
    
    def phase_transition(self):
        betas = np.linspace(0.5, 2.0, 200)
        C = [self.specific_heat(b) for b in betas]
        return betas, C
